// add this near your theme constants, or import from non_btr.dart
enum NonBtrType { houseList, cultivatorList, thandapperNumber, oldSurveyNumber }

extension NonBtrTypeX on NonBtrType {
  String get label {
    switch (this) {
      case NonBtrType.houseList:        return 'House List';
      case NonBtrType.cultivatorList:   return 'Cultivator List';
      case NonBtrType.thandapperNumber: return 'Thandapper Number';
      case NonBtrType.oldSurveyNumber:  return 'Old Survey Number';
    }
  }
  int get btrId {
    switch (this) {
      case NonBtrType.houseList:        return 2;
      case NonBtrType.cultivatorList:   return 3;
      case NonBtrType.thandapperNumber: return 4;
      case NonBtrType.oldSurveyNumber:  return 5;
    }
  }
  IconData get icon {
    switch (this) {
      case NonBtrType.houseList:        return Icons.home_outlined;
      case NonBtrType.cultivatorList:   return Icons.agriculture_outlined;
      case NonBtrType.thandapperNumber: return Icons.format_list_numbered;
      case NonBtrType.oldSurveyNumber:  return Icons.history_edu_outlined;
    }
  }
}

// SEARCHABLE DROPDOWN (generic)
// ═══════════════════════════════════════════════════════════════════════════════
class _SearchableDropdown<T> extends StatelessWidget {
  final T? value;
  final List<T> items;
  final String hint;
  final IconData icon;
  final bool small;
  final String Function(T) displayText;
  final ValueChanged<T?> onChanged;
  final String? Function(T?)? validator;
  final bool enabled;

  const _SearchableDropdown({
    required this.value,
    required this.items,
    required this.hint,
    required this.icon,
    required this.small,
    required this.displayText,
    required this.onChanged,
    this.validator,
    this.enabled = true,
  });

  @override
  Widget build(BuildContext context) {
    final borderRadius = BorderRadius.circular(12);
    final disabledColor = Colors.grey.shade200;
    final fillColor = enabled ? Colors.white : disabledColor;

    return FormField<T>(
      initialValue: value,
      validator: validator,
      builder: (state) {
        return Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            GestureDetector(
              onTap: enabled
                  ? () => _showBottomSheet(context, state)
                  : null,
              child: AnimatedContainer(
                duration: const Duration(milliseconds: 200),
                decoration: BoxDecoration(
                  color: fillColor,
                  borderRadius: borderRadius,
                  border: Border.all(
                    color: state.hasError
                        ? Colors.red.shade400
                        : (value != null ? _secondary : Colors.grey.shade300),
                    width: value != null ? 2 : 1.5,
                  ),
                ),
                padding: EdgeInsets.symmetric(
                    horizontal: small ? 14 : 16,
                    vertical: small ? 13 : 14),
                child: Row(
                  children: [
                    Container(
                      padding: const EdgeInsets.all(7),
                      decoration: BoxDecoration(
                        color: _secondary.withValues(alpha: enabled ? 0.1 : 0.05),
                        borderRadius: BorderRadius.circular(8),
                      ),
                      child: Icon(icon,
                          size: small ? 16 : 18,
                          color: enabled ? _secondary : Colors.grey.shade400),
                    ),
                    SizedBox(width: small ? 10 : 12),
                    Expanded(
                      child: Text(
                        value != null ? displayText(value as T) : hint,
                        style: TextStyle(
                          fontSize: small ? 14 : 15,
                          fontWeight: value != null ? FontWeight.w500 : FontWeight.w400,
                          color: value != null
                              ? Colors.black87
                              : (enabled ? Colors.grey.shade400 : Colors.grey.shade400),
                        ),
                        overflow: TextOverflow.ellipsis,
                      ),
                    ),
                    Icon(
                      Icons.keyboard_arrow_down_rounded,
                      color: enabled ? Colors.grey.shade500 : Colors.grey.shade300,
                      size: 22,
                    ),
                  ],
                ),
              ),
            ),
            if (state.hasError)
              Padding(
                padding: const EdgeInsets.only(top: 6, left: 12),
                child: Text(
                  state.errorText!,
                  style: TextStyle(
                      color: Colors.red.shade700,
                      fontSize: 12,
                      fontWeight: FontWeight.w500),
                ),
              ),
          ],
        );
      },
    );
  }

  void _showBottomSheet(BuildContext context, FormFieldState<T> state) {
    final searchCtrl = TextEditingController();
    final RxList<T> filtered = RxList<T>.from(items);

    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (ctx) => DraggableScrollableSheet(
        initialChildSize: 0.6,
        minChildSize: 0.4,
        maxChildSize: 0.9,
        expand: false,
        builder: (_, scrollController) => Container(
          decoration: const BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
          ),
          child: Column(
            children: [
              // Handle
              Container(
                margin: const EdgeInsets.only(top: 12),
                width: 40,
                height: 4,
                decoration: BoxDecoration(
                  color: Colors.grey.shade300,
                  borderRadius: BorderRadius.circular(2),
                ),
              ),
              // Title
              Padding(
                padding: const EdgeInsets.fromLTRB(20, 16, 20, 8),
                child: Row(
                  children: [
                    Icon(icon, color: _primary, size: 20),
                    const SizedBox(width: 10),
                    Text(
                      'Select $hint'.replaceAll('Select Select ', 'Select '),
                      style: const TextStyle(
                          fontSize: 17,
                          fontWeight: FontWeight.bold,
                          color: Colors.black87),
                    ),
                  ],
                ),
              ),
              // Search bar
              Padding(
                padding: const EdgeInsets.fromLTRB(16, 4, 16, 8),
                child: TextField(
                  controller: searchCtrl,
                  autofocus: true,
                  decoration: InputDecoration(
                    hintText: 'Search…',
                    hintStyle: TextStyle(color: Colors.grey.shade400),
                    prefixIcon: Icon(Icons.search, color: Colors.grey.shade400),
                    filled: true,
                    fillColor: _surface,
                    border: OutlineInputBorder(
                        borderRadius: BorderRadius.circular(12),
                        borderSide: BorderSide.none),
                    contentPadding: const EdgeInsets.symmetric(
                        horizontal: 16, vertical: 12),
                  ),
                  onChanged: (q) {
                    final lower = q.toLowerCase();
                    filtered.value = items
                        .where((item) =>
                        displayText(item).toLowerCase().contains(lower))
                        .toList();
                  },
                ),
              ),
              const Divider(height: 1),
              // List
              Expanded(
                child: Obx(() => filtered.isEmpty
                    ? Center(
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      Icon(Icons.search_off_rounded,
                          size: 40, color: Colors.grey.shade300),
                      const SizedBox(height: 8),
                      Text('No results found',
                          style: TextStyle(color: Colors.grey.shade400)),
                    ],
                  ),
                )
                    : ListView.builder(
                  controller: scrollController,
                  itemCount: filtered.length,
                  itemBuilder: (_, i) {
                    final item = filtered[i];
                    final isSelected = value != null &&
                        displayText(value as T) == displayText(item);
                    return ListTile(
                      leading: Container(
                        padding: const EdgeInsets.all(7),
                        decoration: BoxDecoration(
                          color: isSelected
                              ? _secondary.withValues(alpha: 0.12)
                              : _surface,
                          borderRadius: BorderRadius.circular(8),
                        ),
                        child: Icon(icon,
                            size: 18,
                            color: isSelected ? _secondary : Colors.grey.shade400),
                      ),
                      title: Text(
                        displayText(item),
                        style: TextStyle(
                          fontSize: 15,
                          fontWeight: isSelected
                              ? FontWeight.w700
                              : FontWeight.w500,
                          color: isSelected ? _secondary : Colors.black87,
                        ),
                      ),
                      trailing: isSelected
                          ? const Icon(Icons.check_circle_rounded,
                          color: _secondary, size: 20)
                          : null,
                      onTap: () {
                        onChanged(item);
                        state.didChange(item);
                        Navigator.pop(ctx);
                      },
                    );
                  },
                )),
              ),
              const SizedBox(height: 16),
            ],
          ),
        ),
      ),
    );
  }
}

// AUTOCOMPLETE FIELD (survey no / sub div suggestions)
// ═══════════════════════════════════════════════════════════════════════════════
class _AutocompleteField extends StatefulWidget {
  final TextEditingController controller;
  final List<String> suggestions;
  final String hint;
  final IconData icon;
  final bool small;
  final TextInputType keyboardType;
  final List<TextInputFormatter>? formatters;
  final ValueChanged<String>? onChanged;
  final String? Function(String?)? validator;

  const _AutocompleteField({
    required this.controller,
    required this.suggestions,
    required this.hint,
    required this.icon,
    required this.small,
    this.keyboardType = TextInputType.text,
    this.formatters,
    this.onChanged,
    this.validator,
  });

  @override
  State<_AutocompleteField> createState() => _AutocompleteFieldState();
}

class _AutocompleteFieldState extends State<_AutocompleteField> {
  final _layerLink = LayerLink();
  OverlayEntry? _overlay;
  final _focusNode = FocusNode();

  @override
  void initState() {
    super.initState();
    _focusNode.addListener(() {
      if (!_focusNode.hasFocus) _removeOverlay();
    });
  }

  @override
  void dispose() {
    _removeOverlay();
    _focusNode.dispose();
    super.dispose();
  }

  void _removeOverlay() {
    _overlay?.remove();
    _overlay = null;
  }

  void _showOverlay(List<String> items) {
    _removeOverlay();
    if (items.isEmpty) return;

    final renderBox = context.findRenderObject() as RenderBox?;
    if (renderBox == null) return;
    final size = renderBox.size;

    _overlay = OverlayEntry(
      builder: (_) => Positioned(
        width: size.width,
        child: CompositedTransformFollower(
          link: _layerLink,
          showWhenUnlinked: false,
          offset: Offset(0, size.height + 4),
          child: Material(
            elevation: 8,
            borderRadius: BorderRadius.circular(12),
            color: Colors.white,
            child: ClipRRect(
              borderRadius: BorderRadius.circular(12),
              child: ConstrainedBox(
                constraints: const BoxConstraints(maxHeight: 200),
                child: ListView.builder(
                  padding: EdgeInsets.zero,
                  shrinkWrap: true,
                  itemCount: items.length,
                  itemBuilder: (_, i) => InkWell(
                    onTap: () {
                      widget.controller.text = items[i];
                      widget.onChanged?.call(items[i]);
                      _removeOverlay();
                      _focusNode.unfocus();
                    },
                    child: Padding(
                      padding: const EdgeInsets.symmetric(
                          horizontal: 16, vertical: 12),
                      child: Row(
                        children: [
                          Icon(widget.icon,
                              size: 16, color: _secondary.withValues(alpha: 0.7)),
                          const SizedBox(width: 10),
                          Text(items[i],
                              style: const TextStyle(
                                  fontSize: 14,
                                  fontWeight: FontWeight.w500,
                                  color: Colors.black87)),
                        ],
                      ),
                    ),
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );

    Overlay.of(context).insert(_overlay!);
  }

  @override
  Widget build(BuildContext context) {
    return CompositedTransformTarget(
      link: _layerLink,
      child: TextFormField(
        controller: widget.controller,
        focusNode: _focusNode,
        keyboardType: widget.keyboardType,
        inputFormatters: widget.formatters,
        validator: widget.validator,
        style: TextStyle(
            fontSize: widget.small ? 14 : 15,
            fontWeight: FontWeight.w500,
            color: Colors.black87),
        onChanged: (val) {
          widget.onChanged?.call(val);
          final filtered = widget.suggestions
              .where((s) => s.toLowerCase().contains(val.toLowerCase()))
              .toList();
          if (val.isEmpty) {
            _removeOverlay();
          } else {
            _showOverlay(filtered);
          }
        },
        decoration: InputDecoration(
          hintText: widget.hint,
          hintStyle: TextStyle(
              color: Colors.grey.shade400, fontSize: widget.small ? 13 : 14),
          filled: true,
          fillColor: Colors.white,
          prefixIcon: Container(
            margin: const EdgeInsets.all(8),
            padding: const EdgeInsets.all(7),
            decoration: BoxDecoration(
              color: _secondary.withValues(alpha: 0.1),
              borderRadius: BorderRadius.circular(8),
            ),
            child: Icon(widget.icon,
                size: widget.small ? 16 : 18, color: _secondary),
          ),
          border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide.none),
          enabledBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide:
              BorderSide(color: Colors.grey.shade300, width: 1.5)),
          focusedBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: const BorderSide(color: _secondary, width: 2)),
          errorBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.red.shade400, width: 1.5)),
          focusedErrorBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.red.shade600, width: 2)),
          contentPadding: EdgeInsets.symmetric(
              horizontal: widget.small ? 14 : 16,
              vertical: widget.small ? 13 : 14),
        ),
      ),
    );
  }
}

class _InputField extends StatelessWidget {
  final TextEditingController controller;
  final String hint;
  final IconData icon;
  final bool small;
  final TextInputType keyboardType;
  final int maxLines;
  final List<TextInputFormatter>? formatters;
  final String? Function(String?)? validator;
  final String? suffix;

  const _InputField({
    required this.controller,
    required this.hint,
    required this.icon,
    required this.small,
    this.keyboardType = TextInputType.text,
    this.maxLines     = 1,
    this.formatters,
    this.validator,
    this.suffix,
  });

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: controller,
      keyboardType: keyboardType,
      maxLines: maxLines,
      inputFormatters: formatters,
      validator: validator,
      style: TextStyle(
          fontSize: small ? 14 : 15,
          fontWeight: FontWeight.w500,
          color: Colors.black87),
      decoration: InputDecoration(
        hintText: hint,
        hintStyle: TextStyle(
            color: Colors.grey.shade400, fontSize: small ? 13 : 14),
        filled: true,
        fillColor: Colors.white,
        prefixIcon: Container(
          margin: const EdgeInsets.all(8),
          padding: const EdgeInsets.all(7),
          decoration: BoxDecoration(
            color: _secondary.withValues(alpha: 0.1),
            borderRadius: BorderRadius.circular(8),
          ),
          child: Icon(icon, size: small ? 16 : 18, color: _secondary),
        ),
        suffixText: suffix,
        suffixStyle: TextStyle(
            fontSize: small ? 13 : 14,
            color: _secondary,
            fontWeight: FontWeight.w600),
        border: OutlineInputBorder(
            borderRadius: BorderRadius.circular(12),
            borderSide: BorderSide.none),
        enabledBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(12),
            borderSide:
            BorderSide(color: Colors.grey.shade300, width: 1.5)),
        focusedBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(12),
            borderSide:
            const BorderSide(color: _secondary, width: 2)),
        errorBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(12),
            borderSide:
            BorderSide(color: Colors.red.shade400, width: 1.5)),
        focusedErrorBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(12),
            borderSide:
            BorderSide(color: Colors.red.shade600, width: 2)),
        contentPadding: EdgeInsets.symmetric(
            horizontal: small ? 14 : 16,
            vertical: small ? 13 : 14),
      ),
    );
  }
}

        ),
        child: Column(
          mainAxisSize: MainAxisSize.min,

              style: TextStyle(
